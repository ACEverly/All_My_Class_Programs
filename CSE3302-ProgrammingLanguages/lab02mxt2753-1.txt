const CreateTheHTMLTag=function( beginTag, endTag )
{ 
    return function(textcontent)
    { 
        return beginTag+textcontent+endTag; 
    } 
}

const TableW=CreateTheHTMLTag( '<table>\n','</table>' );
const TrEquals=CreateTheHTMLTag( '<tr>\n','</tr>\n' );
const ThEquals=CreateTheHTMLTag( '<th>','<th>\n' );
const TdEquals=CreateTheHTMLTag('<td>','</td>\n' );
//Let's gwet into the headers
const HeaderTable=TrEquals( ThEquals('Name') + ThEquals('What Popular Band') );
//let's do the tds
const ROne=TrEquals( TdEquals('Adam Levine') + TdEquals('Maroon 5') );
const RTwo=TrEquals( TdEquals('Linkin Park') + TdEquals('Chester Bennington') );
//Now Finally combinbing all of this
const TableOHTML=TableW( HeaderTable+ROne+RTwo );
console.log( TableOHTML );
